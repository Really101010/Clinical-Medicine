// ============================================================
// FITC_fluorescence_batch_simple.ijm
// Batch analyze FITC fluorescence intensity from TIFF images.
// Records results by original filename only.
// ============================================================

// ---------- Parameters ----------
channelKeywords = "green,gfp,fitc,egfp"; // search in channel titles
autoThresholdMethod = "Otsu dark"; // "Otsu dark", "Default dark", "Triangle dark"
// --------------------------------

inputDir = getDirectory("Choose the input folder containing TIFF images");
outputDir = getDirectory("Choose the output folder for the results CSV");

fileList = getFileList(inputDir);

run("Clear Results");
setBatchMode(true);

count = 0;

for (i = 0; i < fileList.length; i++) {
    fileName = fileList[i];
    lowerName = toLowerCase(fileName);

    // Only process TIFF files
    if (endsWith(lowerName, ".tif") || endsWith(lowerName, ".tiff")) {
        path = inputDir + fileName;
        open(path);

        // Check dimensions
        getDimensions(width, height, channels, slices, frames);

        // Handle multi-channel / RGB images
        if (channels > 1 || bitDepth() == 24) {
            run("Split Channels");
            titles = getList("image.titles");
            chosen = "";
            keywords = split(channelKeywords, ",");
            
            for (j = 0; j < titles.length; j++) {
                t = toLowerCase(titles[j]);
                for (k = 0; k < keywords.length; k++) {
                    if (indexOf(t, keywords[k]) >= 0) {
                        chosen = titles[j];
                    }
                }
            }
            
            if (chosen == "") {
                print("WARNING: No FITC/green channel found in " + fileName + ", skipping.");
                closeAllImages();
                continue;
            }
            selectWindow(chosen);
        }
        // Else: single-channel TIFF, assume it is already FITC channel

        // Optional: subtract background (uncomment if needed)
        // run("Subtract Background...", "rolling=50");

        // Set auto threshold
        setAutoThreshold(autoThresholdMethod);
        setOption("BlackBackground", true);
        
        // Measure only thresholded pixels
        run("Set Measurements...", "area mean min max integrated limit redirect=None decimal=3");
        run("Measure");

        // Add filename to the last row
        row = nResults - 1;
        setResult("Filename", row, fileName);
        updateResults();

        count++;
        closeAllImages();
    }
}

setBatchMode(false);

// Save results
if (count > 0) {
    outPath = outputDir + "FITC_measure_results.csv";
    if (File.exists(outPath)) {
        File.delete(outPath);
    }
    saveAs("Results", outPath);
    print("Done. Processed " + count + " file(s). Results saved to: " + outPath);
} else {
    print("No matching TIFF files found.");
}

// ---------- Helper: close all images ----------
function closeAllImages() {
    while (nImages > 0) {
        selectImage(nImages);
        close();
    }
}

// ============================================================
// Green_Red_whole_image_batch.ijm
// Batch measure green & red fluorescence intensity from merged TIFF images.
// 修改说明：采用全图测量法，取消自动阈值，直接测量整图的 Mean 和 IntDen。
// ============================================================

// ---------- Parameters ----------
greenKeywords = "green,gfp,egfp,fitc,alexa488,cy2";
redKeywords   = "red,rfp,mcherry,dsred,tritc,cy3,alexa555,alexa594,pi,propidium";

analyzeGreen = true;
analyzeRed   = true;

// 关键词匹配失败时的回退通道号（1-based）
// RGB 拆分顺序固定为 C1=Red, C2=Green, C3=Blue
fallbackGreenChannel = 2;
fallbackRedChannel   = 1;

// 单通道 TIFF 无法自动判断红/绿，这里指定角色："Green" / "Red" / "Unknown"
singleChannelRole = "Green";

// --------------------------------

inputDir  = getDirectory("Choose the input folder containing TIFF images");
outputDir = getDirectory("Choose the output folder for the results CSV");

fileList = getFileList(inputDir);

run("Clear Results");
setBatchMode(true);

count = 0;

for (i = 0; i < fileList.length; i++) {
    fileName  = fileList[i];
    lowerName = toLowerCase(fileName);

    if (!(endsWith(lowerName, ".tif") || endsWith(lowerName, ".tiff"))) continue;

    open(inputDir + fileName);
    if (nImages == 0) continue;

    getDimensions(width, height, channels, slices, frames);
    wasRGB = (bitDepth() == 24);

    if (channels > 1 || wasRGB) {
        origTitle = getTitle();
        run("Split Channels");

        allTitles = getList("image.titles");
        chTitles  = newArray(0);
        for (j = 0; j < allTitles.length; j++) {
            if (allTitles[j] != origTitle) {
                chTitles = Array.concat(chTitles, allTitles[j]);
            }
        }

        greenTitle = findChannelTitle(chTitles, greenKeywords);
        redTitle   = findChannelTitle(chTitles, redKeywords);

        if (greenTitle == "" && analyzeGreen && fallbackGreenChannel > 0)
            greenTitle = channelByIndex(chTitles, fallbackGreenChannel);
        if (redTitle == "" && analyzeRed && fallbackRedChannel > 0)
            redTitle = channelByIndex(chTitles, fallbackRedChannel);

        if (analyzeGreen) {
            if (greenTitle != "") {
                measureChannel(greenTitle, fileName, "Green");
                count++;
            } else print("WARNING: no green channel in " + fileName);
        }
        if (analyzeRed) {
            if (redTitle != "") {
                measureChannel(redTitle, fileName, "Red");
                count++;
            } else print("WARNING: no red channel in " + fileName);
        }

    } else {
        // 单通道
        if (singleChannelRole == "Green" && analyzeGreen) {
            measureChannel(getTitle(), fileName, "Green");
            count++;
        } else if (singleChannelRole == "Red" && analyzeRed) {
            measureChannel(getTitle(), fileName, "Red");
            count++;
        } else {
            print("WARNING: single-channel image with unknown role, skipping: " + fileName);
        }
    }

    closeAllImages();
}

setBatchMode(false);

if (count > 0) {
    outPath = outputDir + "WholeImage_Fluorescence_results.csv";
    if (File.exists(outPath)) File.delete(outPath);
    saveAs("Results", outPath);
    print("Done. " + count + " measurement(s). Saved to: " + outPath);
} else {
    print("No matching TIFF files / channels found.");
}

// ---------- Helpers ----------
function findChannelTitle(titles, keywordCsv) {
    kws  = split(keywordCsv, ",");
    best = "";
    for (j = 0; j < titles.length; j++) {
        t = toLowerCase(titles[j]);
        for (k = 0; k < kws.length; k++) {
            if (indexOf(t, kws[k]) >= 0) best = titles[j]; 
        }
    }
    return best;
}

function channelByIndex(titles, idx) {
    target = "c" + idx + "-";
    for (j = 0; j < titles.length; j++) {
        if (indexOf(toLowerCase(titles[j]), target) == 0) return titles[j];
    }
    if (idx >= 1 && idx <= titles.length) return titles[idx - 1];
    return "";
}

function measureChannel(title, fileName, role) {
    selectWindow(title);
    
    // ==========================================
    // 方案二核心修改：
    // 1. 去掉了 setAutoThreshold 自动阈值
    // 2. 去掉了 setOption("BlackBackground", true)
    // 3. 测量参数只保留 mean（均值）, min, max 和 integrated（总荧光量）
    // ==========================================
    run("Set Measurements...", "mean min max integrated redirect=None decimal=3");
    run("Measure");
    
    row = nResults - 1;
    setResult("Filename", row, fileName);
    setResult("Channel",  row, role);
    setResult("Title",    row, title);
    updateResults();
}

function closeAllImages() {
    while (nImages > 0) {
        selectImage(nImages);
        close();
    }
}
