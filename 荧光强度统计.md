```
// ============================================================
// FITC_fluorescence_batch_simple.ijm
// Batch analyze FITC fluorescence intensity from TIFF images.
// Records results by original filename only.
// ============================================================

// ---------- Parameters ----------
channelKeywords = "green,gfp,fitc,egfp"; // search in channel titles
autoThresholdMethod = "Otsu dark"; // "Otsu dark", "Default dark", "Triangle dark"
// --------------------------------

inputDir = getDirectory("Choose the input folder containing TIFF images");
outputDir = getDirectory("Choose the output folder for the results CSV");

fileList = getFileList(inputDir);

run("Clear Results");
setBatchMode(true);

count = 0;

for (i = 0; i < fileList.length; i++) {
    fileName = fileList[i];
    lowerName = toLowerCase(fileName);

    // Only process TIFF files
    if (endsWith(lowerName, ".tif") || endsWith(lowerName, ".tiff")) {
        path = inputDir + fileName;
        open(path);

        // Check dimensions
        getDimensions(width, height, channels, slices, frames);

        // Handle multi-channel / RGB images
        if (channels > 1 || bitDepth() == 24) {
            run("Split Channels");
            titles = getList("image.titles");
            chosen = "";
            keywords = split(channelKeywords, ",");
            
            for (j = 0; j < titles.length; j++) {
                t = toLowerCase(titles[j]);
                for (k = 0; k < keywords.length; k++) {
                    if (indexOf(t, keywords[k]) >= 0) {
                        chosen = titles[j];
                    }
                }
            }
            
            if (chosen == "") {
                print("WARNING: No FITC/green channel found in " + fileName + ", skipping.");
                closeAllImages();
                continue;
            }
            selectWindow(chosen);
        }
        // Else: single-channel TIFF, assume it is already FITC channel

        // Optional: subtract background (uncomment if needed)
        // run("Subtract Background...", "rolling=50");

        // Set auto threshold
        setAutoThreshold(autoThresholdMethod);
        setOption("BlackBackground", true);
        
        // Measure only thresholded pixels
        run("Set Measurements...", "area mean min max integrated limit redirect=None decimal=3");
        run("Measure");

        // Add filename to the last row
        row = nResults - 1;
        setResult("Filename", row, fileName);
        updateResults();

        count++;
        closeAllImages();
    }
}

setBatchMode(false);

// Save results
if (count > 0) {
    outPath = outputDir + "FITC_measure_results.csv";
    if (File.exists(outPath)) {
        File.delete(outPath);
    }
    saveAs("Results", outPath);
    print("Done. Processed " + count + " file(s). Results saved to: " + outPath);
} else {
    print("No matching TIFF files found.");
}

// ---------- Helper: close all images ----------
function closeAllImages() {
    while (nImages > 0) {
        selectImage(nImages);
        close();
    }
}
```
