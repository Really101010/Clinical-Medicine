```
// 批处理宏：添加标尺、Flatten、保存
// 请根据你的实际需求修改标尺参数

inputDir = getDirectory("Choose the input directory"); // 选择原始图片文件夹
outputDir = getDirectory("Choose the output directory"); // 选择输出文件夹

list = getFileList(inputDir); // 获取文件列表
Array.sort(list); // 排序

for (i = 0; i < list.length; i++) {
    filename = list[i];
    if (endsWith(filename, ".tif") || endsWith(filename, ".jpg") || endsWith(filename, ".png")) { // 只处理指定格式
        open(inputDir + filename); // 打开图片

        // 添加标尺
        run("Scale Bar...", "width=300 height=20 font=60 color=White background=None location=[Lower Right] bold overlay"); // 可改width即标尺长度

        // Flatten（将标尺烧录到像素中）
        run("Flatten");

        // 保存图片
        saveAs("Tiff", outputDir + filename); // 可改为 "Jpeg" 或 "Png"

        close(); // 关闭当前图片窗口
    }
}
print("批量处理完成！");
```
